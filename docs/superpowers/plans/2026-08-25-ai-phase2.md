# In src/services/store.py, add:

def get_active_faqs(db: Session) -> list[dict[str, Any]]:
    """Get all active FAQs ordered by order_index."""
    stmt = text("""
        SELECT faq_id, question, answer, category, view_count, order_index
        FROM faq
        WHERE is_active = TRUE
        ORDER BY order_index ASC, faq_id ASC
    """)
    result = db.execute(stmt)
    return [dict(row._mapping) for row in result.fetchall()]


def get_all_faqs(db: Session) -> list[dict[str, Any]]:
    """Get all FAQs (including inactive) for admin."""
    stmt = text("""
        SELECT faq_id, question, answer, category, is_active, view_count, order_index,
               created_by
        FROM faq
        ORDER BY order_index ASC, faq_id ASC
    """)
    result = db.execute(stmt)
    return [dict(row._mapping) for row in result.fetchall()]


def create_faq(
    db: Session,
    question: str,
    answer: str,
    created_by: int,
    category: str | None = None,
) -> dict[str, Any]:
    """Create a new FAQ."""
    stmt = text("""
        INSERT INTO faq (question, answer, created_by, category, is_active, order_index)
        VALUES (:question, :answer, :created_by, :category, TRUE, 0)
        RETURNING faq_id, question, answer, category, is_active, view_count, order_index
    """)
    result = db.execute(stmt, {
        "question": question,
        "answer": answer,
        "created_by": created_by,
        "category": category,
    })
    db.commit()
    return dict(result.one()._mapping)


def update_faq(
    db: Session,
    faq_id: int,
    question: str | None = None,
    answer: str | None = None,
    category: str | None = None,
    is_active: bool | None = None,
    order_index: int | None = None,
) -> dict[str, Any]:
    """Update an FAQ."""
    updates = []
    params = {"faq_id": faq_id}
    
    if question is not None:
        updates.append("question = :question")
        params["question"] = question
    if answer is not None:
        updates.append("answer = :answer")
        params["answer"] = answer
    if category is not None:
        updates.append("category = :category")
        params["category"] = category
    if is_active is not None:
        updates.append("is_active = :is_active")
        params["is_active"] = is_active
    if order_index is not None:
        updates.append("order_index = :order_index")
        params["order_index"] = order_index

    if not updates:
        return get_faq_by_id(db, faq_id)

    stmt = text(f"""
        UPDATE faq
        SET {', '.join(updates)}
        WHERE faq_id = :faq_id
        RETURNING faq_id, question, answer, category, is_active, view_count, order_index
    """)
    result = db.execute(stmt, params)
    db.commit()
    return dict(result.one()._mapping)


def delete_faq(db: Session, faq_id: int) -> bool:
    """Delete an FAQ."""
    stmt = text("DELETE FROM faq WHERE faq_id = :faq_id")
    result = db.execute(stmt, {"faq_id": faq_id})
    db.commit()
    return result.rowcount > 0


def increment_faq_view_count(db: Session, faq_id: int) -> None:
    """Increment view count when FAQ is shown."""
    stmt = text("UPDATE faq SET view_count = view_count + 1 WHERE faq_id = :faq_id")
    db.execute(stmt, {"faq_id": faq_id})
    db.commit()


def get_faq_by_id(db: Session, faq_id: int) -> dict[str, Any] | None:
    """Get FAQ by ID."""
    stmt = text("SELECT * FROM faq WHERE faq_id = :faq_id")
    result = db.execute(stmt, {"faq_id": faq_id})
    row = result.fetchone()
    return dict(row._mapping) if row else None
```

# src/agents/nodes/faq_match.py
import logging
from typing import Any

from src.database import SessionLocal
from src.services import store

logger = logging.getLogger(__name__)

# Simple keyword matching threshold
FAQ_KEYWORD_THRESHOLD = 0.3  # At least 30% keyword overlap


def _keyword_similarity(query: str, faq_question: str) -> float:
    """Calculate keyword overlap between query and FAQ question.
    
    Returns a score between 0 and 1.
    """
    query_words = set(query.lower().split())
    question_words = set(faq_question.lower().split())
    
    # Remove common stop words
    stop_words = {
        "là", "của", "và", "có", "được", "cho", "với", "trong",
        "không", "để", "tôi", "muốn", "hỏi", "về", "cần", "phải",
        "theo", "nào", "bao", "giờ", "ở", "đây", "ra", "vào",
        "a", "an", "the", "is", "are", "do", "does", "what", "how", "when"
    }
    
    query_words = query_words - stop_words
    question_words = question_words - stop_words
    
    if not query_words:
        return 0.0
    
    intersection = query_words & question_words
    union = query_words | question_words
    
    return len(intersection) / len(union) if union else 0.0


async def find_matching_faq(query: str) -> dict[str, Any] | None:
    """Find best matching FAQ for a query.
    
    Returns the FAQ with highest keyword similarity if above threshold,
    otherwise returns None.
    """
    db = SessionLocal()
    try:
        faqs = store.get_active_faqs(db)
        
        best_match = None
        best_score = 0.0
        
        for faq in faqs:
            score = _keyword_similarity(query, faq["question"])
            if score > best_score:
                best_score = score
                best_match = faq
        
        if best_score >= FAQ_KEYWORD_THRESHOLD and best_match:
            # Increment view count
            store.increment_faq_view_count(db, best_match["faq_id"])
            return best_match
        
        return None
    finally:
        db.close()


def build_faq_response(faq: dict[str, Any]) -> dict[str, Any]:
    """Build agent state update from FAQ match."""
    return {
        "answer": faq["answer"],
        "answer_source": "faq",
        "confidence": 0.95,  # FAQ is curated, high confidence
        "sources": [],
        "matched_faq_id": faq["faq_id"],
        "matched_faq_question": faq["question"],
    }
```

# In src/agents/graph.py, modify the graph structure:

from src.agents.nodes.faq_match import find_matching_faq, build_faq_response

# Define FAQ node
def faq_node(state: AgentState) -> AgentState:
    """Check FAQ before RAG. Returns FAQ answer if match found."""
    query = state.get("query", "")
    
    # Run FAQ matching (sync)
    import asyncio
    faq = asyncio.run(find_matching_faq(query))
    
    if faq:
        logger.info(f"FAQ match found: {faq['faq_id']}")
        return build_faq_response(faq)
    
    # No FAQ match → continue to RAG
    logger.info("No FAQ match, continuing to RAG")
    return {"next_node": "rag"}


# Add FAQ node to graph
graph_builder.add_node("faq", faq_node)

# Add conditional edge from faq to end or rag
def faq_routing(state: AgentState) -> str:
    """Route from FAQ node."""
    if state.get("answer_source") == "faq":
        return "end"
    return "rag"

graph_builder.add_conditional_edges(
    "faq",
    faq_routing,
    {
        "end": END,
        "rag": "rag_retrieve",
    }
)

# Modify START to go to faq first
graph_builder.add_edge(START, "faq")

async def suggest_ticket_creation(query: str) -> dict[str, Any]:
    """Return state update suggesting ticket creation."""
    return {
        "answer": TICKET_SUGGESTION_MESSAGE,
        "answer_source": "no_context",
        "confidence": 0.0,
        "sources": [],
        "ticket_suggested": True,
        "ticket_query": query,

async def create_ticket_from_chat(
    user_id: int,
    query: str,
) -> dict[str, Any] | None:
    """Create a support ticket from chat query.
    
    Called when user explicitly asks to create a ticket,
    or when system auto-detects need for human assistance.
    """
    db = SessionLocal()
    try:
        # Generate subject from query
        subject = query[:200] if len(query) > 200 else query
        if not subject:
            subject = "Câu hỏi từ chat"

        # Create ticket with initial message
        ticket = store.create_ticket(
            db,
            user_id=user_id,
            subject=subject,
            initial_message=query,
        )

        logger.info(f"Ticket created from chat: {ticket['ticket_id']} for user {user_id}")
        return ticket
    except Exception as e:
        logger.error(f"Failed to create ticket from chat: {e}")
        return None
    finally:
        db.close()

from src.services.ticket_creation import (
    NO_CONTEXT_MESSAGE,
    LOW_CONFIDENCE_MESSAGE,
    suggest_ticket_creation,
)

RAG_MAX_DISTANCE = 0.35  # Threshold for low confidence


async def rag_retrieve_node(state: AgentState) -> AgentState:
    """RAG retrieval with ticket creation fallback."""
    query = state.get("query", "")
    retrieved = await rag_retrieve(query)
    
    # No documents found
    if retrieved["count"] == 0:
        logger.info("RAG: no documents found, suggesting ticket")
        return await suggest_ticket_creation(query)
    
    # Low confidence
    max_distance = retrieved["max_distance"]
    if max_distance > RAG_MAX_DISTANCE:
        logger.info(f"RAG: low confidence ({max_distance:.3f} > {RAG_MAX_DISTANCE})")
        return {
            "answer": LOW_CONFIDENCE_MESSAGE,
            "answer_source": "no_context",
            "confidence": 1 - max_distance,
            "sources": retrieved.get("sources", []),
            "ticket_suggested": True,
            "ticket_query": query,
        }
    
    # Normal RAG flow
    return {
        "sources": retrieved.get("sources", []),
        "max_distance": max_distance,
        "retrieved_count": retrieved["count"],
    }
